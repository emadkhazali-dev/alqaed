async function fetchData() { const tbody = document.getElementById('table-body'); try { const response = await fetch(SCRIPT_URL + '?nocache=' + new Date().getTime()); const data = await response.json(); tbody.innerHTML = ""; let totalDebt = 0; data.forEach(row => { // تحويل القيم إلى أرقام بدقة const agraValue = parseFloat(row.agra) || 0; const prodCommValue = parseFloat(row.commProduct) || 0; const grandTotal = agraValue + prodCommValue; totalDebt += grandTotal; const msg = `شركة القائد:\nكابتن ${row.name}\nالأجرة: ${agraValue}\nعمولة المنتج: ${prodCommValue.toFixed(2)}\nالمجموع المطلوب: ${grandTotal.toFixed(2)} دينار.`; const waLink = `https://wa.me/${row.phone}?text=${encodeURIComponent(msg)}`; tbody.innerHTML += ` ${row.name} ${agraValue} JOD ${prodCommValue.toFixed(2)} ${grandTotal.toFixed(2)} دينار تنبيه واتساب ✅ `; }); document.getElementById('total-debt').innerText = `إجمالي المديونية: ${totalDebt.toFixed(2)} دينار`; } catch (e) { console.log("خطأ في جلب البيانات"); } }
